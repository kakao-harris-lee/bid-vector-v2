import bidvector.sharedkernel.Fact
import bidvector.strategy.FullScopeText
import bidvector.strategy.KeywordScopeText
import bidvector.strategy.WatchSubject

fun positiveKeywordScopeOwnText() {
    val keyword = KeywordScopeText("키워드 텍스트")
    val full = FullScopeText("전체 텍스트")
    WatchSubject(emptySet(), keyword, full, Fact.Absent(bidvector.sharedkernel.ReasonCode.EMPTY_INPUT))
}

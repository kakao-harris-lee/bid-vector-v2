import bidvector.sharedkernel.Fact
import bidvector.strategy.WatchSubject
import bidvector.strategy.assembleFullScopeText
import bidvector.strategy.assembleKeywordScopeText

fun positiveKeywordScopeOwnText() {
    val keyword = assembleKeywordScopeText("키워드 텍스트", null)
    val full = assembleFullScopeText("전체 텍스트", null, null, null)
    WatchSubject(emptySet(), keyword, full, Fact.Absent(bidvector.sharedkernel.ReasonCode.EMPTY_INPUT))
}

fun appliedFloorArgument(
    base: BaseAmount,
    rate: BidRate,
    policy: Resolution.Resolved<RoundingPolicy>,
): Measurement<Derived<BidAmount>> = (base * rate).roundedWith(policy, base.appliedFloorFrom(rate))
